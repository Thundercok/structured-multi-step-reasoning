"""tests.test_rat_completion_reconciliation — Headless fixtures verifying ADR-003 reconciliation contract.

Covers:
1. Omitted outcome (model claims saved/done in prose without executing tool)
2. Wrong-source producer/consumer (calculate 324, save 126 is rejected)
3. Cross-execution isolation (receipt from old execution cannot satisfy new mutation)
4. Contract freeze and downgrade prevention (frozen obligations cannot be removed)
5. Pending input lifecycle (waiting InputRequest records needs_input and unresolved)
6. Two separate cancel fixtures:
   - Cancel after calculate commit: preserves committed artifact, blocks late writes, resumes via new execution
   - Cancel during waiting probe: 0 artifacts committed, records cancelled and unresolved
7. Pure conversational response without tools (satisfies response contract without faking quality PASS)
8. Unconfirmed proposal does not satisfy confirmed obligation
9. Boundary removal (note created then removed in same execution is unmet)
10. Stale revision update (wrong expected revision is unmet)
"""
from unittest.mock import MagicMock
import pytest

from rat.engine.assistant_runtime import AssistantRuntime, ModelDecision
from rat.engine.completion import (
    ExecutionJournal,
    OutcomeKind,
    OutcomeSpec,
    ReceiptEnvelope,
    TurnContract,
    TurnReconciliation,
    reconcile_turn,
)
from rat.kernel.contracts import Artifact, InputRequest
from rat.timetable.compositor import TimetableCompositor
from rat.ui.chat_session import ConversationHistory


def runtime_fixture():
    return AssistantRuntime(db=MagicMock(), compositor=TimetableCompositor([]),
                            use_live_model=False, plan_tasks=False)


def action(name, **args):
    return ModelDecision("", tool_name=name, tool_args=args)


def answer(text, **kwargs):
    return ModelDecision("", final_answer=text, **kwargs)


def result_note(artifact, text, **replacement):
    return action("record_work_note", text=text, source_result={
        "artifact_id": artifact.id, "revision": artifact.revision, "field": "result",
    }, **replacement)


def run_turn(runtime, history, query, decisions, contract=None, **options):
    prior_messages = history.messages_with_turn_ids()
    prior_refs = history.turn_references()
    snap = history.begin_turn(query)
    choices = iter(decisions)
    def model_callable(goal, steps):
        choice = next(choices)
        return choice() if callable(choice) else choice
    runtime.model_callable = model_callable
    response = runtime.run(
        query,
        history=prior_messages,
        turn_references=prior_refs,
        turn_ref=snap.turn_ref,
        execution_id=snap.execution_id,
        conversation_id=history.conversation_id,
        contract=contract,
        **options,
    )
    refs = [{"artifact_id": a.id, "revision": a.revision} for a in response.artifacts]
    metadata = dict(
        execution_id=response.execution_id,
        artifact_refs=refs,
        note_ids=list(dict.fromkeys(e["note_id"] for e in response.note_events)),
        note_event_ids=[e["event_id"] for e in response.note_events],
    )
    if response.error == "cancelled":
        history.cancel_turn(snap.turn_id, reconciliation=response.reconciliation, **metadata)
    elif response.error:
        history.fail_turn(snap.turn_id, error=response.error, reconciliation=response.reconciliation, **metadata)
    else:
        history.complete_turn(snap.turn_id, response.answer, reconciliation=response.reconciliation, **metadata)
    return snap, response

# 1. OMITTED OUTCOME: MODEL PROMISES SAVED WITHOUT EXECUTING TOOL
# =============================================================================

def test_omitted_outcome_model_claims_saved_without_tool_execution():
    runtime, history = runtime_fixture(), ConversationHistory()
    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-save", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note", description="Ghi lại kết quả"),
            OutcomeSpec("out-resp", kind=OutcomeKind.RESPONSE, description="Trả lời xác nhận"),
        ],
        intent_coverage="model_proposed",
    )
    snap, response = run_turn(
        runtime, history, "Ghi lại kết quả giúp tôi.",
        [answer("Tôi đã ghi lại kết quả vào sổ tay rồi nhé.")],
        contract=contract,
    )

    assert response.error is None
    assert len(response.note_events) == 0
    assert response.completion_status == "unresolved"
    assert response.reconciliation is not None
    assert response.reconciliation.declared_scope_completion == "unresolved"
    assert any(u["outcome_id"] == "out-save" and u["reason"] == "omitted_action" for u in response.reconciliation.unmet_outcomes)

    # UI / History wiring verification
    turn_record = history.get_turn(snap.turn_id)
    assert turn_record.reconciliation_status == "unresolved"
    assert "omitted_action" in turn_record.reconciliation["unresolved_reasons"]
    latest_ref = history.turn_references()[-1]
    assert latest_ref["reconciliation_status"] == "unresolved"
    assert "omitted_action" in latest_ref["unresolved_reasons"]

# 2. WRONG-SOURCE PRODUCER/CONSUMER: CALCULATE 324 BUT SAVE 126
# =============================================================================

def test_wrong_source_producer_consumer_calculate_324_save_126_fails():
    runtime, history = runtime_fixture(), ConversationHistory()

    # Step 1: establish old artifact 126
    _, first = run_turn(runtime, history, "Tính 18 * 7", [
        action("calculate", expression="18 * 7"), answer("126"),
    ])
    old_artifact = first.artifacts[0]
    assert old_artifact.payload["result"] == "126"

    # Step 2: request calculate double AND record that newly calculated result
    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-calc", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="calculate", description="Tính 162 * 2"),
            OutcomeSpec("out-note", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note",
                        source_ref={"producer_outcome_id": "out-calc"}, dependency_ids=["out-calc"],
                        description="Ghi kết quả vừa tính"),
            OutcomeSpec("out-ans", kind=OutcomeKind.RESPONSE, description="Trả lời"),
        ],
        intent_coverage="model_proposed",
    )

    # Model calculates 162 * 2 (producing 324), but records the old artifact 126!
    _, response = run_turn(
        runtime, history, "Tính 162 * 2 rồi ghi lại kết quả vừa tính.",
        [
            action("calculate", expression="162 * 2"),
            result_note(old_artifact, "126"),
            answer("Đã tính và ghi lại."),
        ],
        contract=contract,
    )

    assert response.error is None
    # Calculate succeeded, but note consumer pointed to wrong producer source
    assert response.completion_status == "partial"
    assert response.reconciliation.declared_scope_completion == "partial"
    unmet = response.reconciliation.unmet_outcomes
    assert len(unmet) == 1
    assert unmet[0]["outcome_id"] == "out-note"
    assert unmet[0]["reason"] == "wrong_source_consumer"

# 3. CROSS-EXECUTION ISOLATION: OLD EXECUTION RECEIPTS CANNOT SATISFY NEW MUTATION
# =============================================================================

def test_cross_execution_receipt_cannot_satisfy_new_mutation():
    runtime, history = runtime_fixture(), ConversationHistory()
    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-note", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note"),
        ],
        intent_coverage="model_proposed",
    )

    # Create fake journal containing an old receipt from previous execution
    old_journal = ExecutionJournal("conv-1", "turn-1", "exec-old", 1)
    old_journal.receipts.append(ReceiptEnvelope(
        receipt_id="rcpt-old-note",
        issuer="notebook",
        conversation_id="conv-1",
        turn_id="turn-1",
        execution_id="exec-old",
        operation_name="record_work_note",
        status="committed",
        committed=True,
        note_event={"owner_turn_id": "turn-1", "context": {"execution_id": "exec-old"}, "snapshot": {"confirmed": True}},
    ))

    # Reconcile for new execution 'exec-new'
    reconciliation = reconcile_turn(
        contract,
        old_journal,
        lifecycle="completed",
        final_answer="Done",
    )

    assert reconciliation.declared_scope_completion == "unresolved"
    assert any(u["outcome_id"] == "out-note" and u["reason"] == "omitted_action" for u in reconciliation.unmet_outcomes)

# 4. CONTRACT FREEZE: DOWNGRADE PREVENTION AFTER FIRST EFFECT
# =============================================================================

def test_contract_freeze_prevents_downgrade_or_obligation_deletion():
    runtime, history = runtime_fixture(), ConversationHistory()
    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-calc", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="calculate"),
            OutcomeSpec("out-note", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note"),
        ],
        intent_coverage="model_proposed",
    )

    # Model executes calculate, then tries to return a decision with contract having 0 outcomes (downgrade)
    downgraded_decision = answer("Only calculated, deleting note obligation")
    downgraded_decision.contract = TurnContract(outcomes=[], intent_coverage="model_proposed")

    _, response = run_turn(
        runtime, history, "Tính và ghi",
        [
            action("calculate", expression="10 + 5"),
            downgraded_decision,
        ],
        contract=contract,
    )

    # Frozen contract was preserved; note remains unmet
    assert response.reconciliation.declared_scope_completion == "partial"
    assert any(u["outcome_id"] == "out-note" and u["reason"] == "omitted_action" for u in response.reconciliation.unmet_outcomes)

def test_pending_input_reconciles_to_needs_input_and_unresolved():
    runtime, history = runtime_fixture(), ConversationHistory()
    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-1", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="custom_input_op"),

        intent_coverage="model_proposed",
    )

    # Simulate pending input request in dispatcher

    # Decision returning needs_input

    assert "pending_input_required" in response.reconciliation.unresolved_reasons

# 6. TWO SEPARATE CANCEL FIXTURES:
#    A) Cancel after calculate commit: preserves artifact, blocks late writes,
#       and resumes via new execution
#    B) Cancel during waiting probe: 0 artifacts committed
# =============================================================================

def test_cancel_after_calculate_commit_preserves_artifact_and_resumes_in_new_exec():
    runtime, history = runtime_fixture(), ConversationHistory()

    # Step 1: calculate and record
    _, t1 = run_turn(runtime, history, "Tính 18 * 9", [
        action("calculate", expression="18 * 9"),
        lambda: result_note(runtime.artifact_store.list("calculation")[-1], "162"),
        answer("162"),
    ])
    note_id = t1.note_events[0]["note_id"]

    # Step 2: calculate double (324) and cancel immediately after calculate commit
    cancelled_flag = {"cancelled": False}
    def trigger_cancel_after_calc(step):
        if step.tool_name == "calculate":
            cancelled_flag["cancelled"] = True

    contract_t2 = TurnContract(
        outcomes=[
            OutcomeSpec("out-calc", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="calculate"),
            OutcomeSpec("out-note", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note"),
        ],
        intent_coverage="model_proposed",
    )

    snap_stopped, stopped = run_turn(
        runtime, history, "Tính gấp đôi rồi ghi lại.",
        [
            action("calculate", expression="162 * 2"),
            answer("Unreachable committed text"),
        ],
        contract=contract_t2,
        on_step=trigger_cancel_after_calc,
        should_cancel=lambda: cancelled_flag["cancelled"],
    )

    assert stopped.error == "cancelled"
    assert stopped.reconciliation.lifecycle == "cancelled"
    assert stopped.reconciliation.declared_scope_completion == "unresolved"
    # Committed artifact 324 is PRESERVED in store
    assert len(stopped.artifacts) == 1
    artifact_324 = stopped.artifacts[0]
    assert artifact_324.payload["result"] == "324"

    # Late-write guard: attempting to record note on stopped execution is rejected
    stopped_ctx = runtime.notebook.current_context()
    with pytest.raises(ValueError, match="cancelled"):
        runtime.execute_tool(
            "record_work_note",
            result_note(artifact_324, "324").tool_args,
            turn_id=snap_stopped.turn_id,
            execution_id=stopped.execution_id,
            note_context=stopped_ctx,
        )

    # Turn 6 in history remains cancelled
    assert history.get_turn(snap_stopped.turn_id).status == "cancelled"

    # Step 3: Resume in a NEW execution, referencing the preserved artifact 324
    contract_t3 = TurnContract(
        outcomes=[
            OutcomeSpec("out-inspect", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="inspect_artifact"),
            OutcomeSpec("out-save", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note"),
            OutcomeSpec("out-resp", kind=OutcomeKind.RESPONSE),
        ],
        intent_coverage="model_proposed",
    )

    snap_resumed, resumed = run_turn(
        runtime, history, "Tiếp tục, ghi kết quả vừa tính.",
        [
            action("inspect_artifact", artifact_id=artifact_324.id, revision=1),
            result_note(artifact_324, "324", replace_note_id=note_id, revision=1),
            answer("Đã ghi kết quả vừa tính."),
        ],
        contract=contract_t3,
    )

    assert resumed.error is None
    assert resumed.execution_id != stopped.execution_id
    assert resumed.reconciliation.lifecycle == "completed"
    assert resumed.reconciliation.declared_scope_completion == "satisfied"
    assert len(resumed.note_events) == 1
    assert resumed.note_events[0]["snapshot"]["text"] == "324"
    assert history.get_turn(snap_stopped.turn_id).status == "cancelled"
    assert history.get_turn(snap_resumed.turn_id).status == "completed"


def test_cancel_during_waiting_probe_has_zero_artifacts():
    runtime, history = runtime_fixture(), ConversationHistory()
    contract = TurnContract(
        outcomes=[OutcomeSpec("out-calc", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="calculate")],
        intent_coverage="model_proposed",
    )

    # Cancel triggered before any tool starts
    snap, response = run_turn(
        runtime, history, "Tính phép toán lớn",
        [action("calculate", expression="1000 ** 2")],
        contract=contract,
        should_cancel=lambda: True,
    )

    assert response.error == "cancelled"
    assert response.reconciliation.lifecycle == "cancelled"
    assert response.reconciliation.declared_scope_completion == "unresolved"
    assert len(response.artifacts) == 0

# 8. UNCONFIRMED PROPOSAL DOES NOT SATISFY CONFIRMED OBLIGATION
# =============================================================================

def test_proposed_note_does_not_satisfy_confirmed_obligation():
    runtime, history = runtime_fixture(), ConversationHistory()
    runtime.artifact_store.put(Artifact("sample", "sample", 1, {"result": "42"}))
    ref = {"artifact_id": "sample", "revision": 1, "field": "result"}

    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-note", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note",
                        allow_proposal=False, description="Ghi kết quả đã kiểm chứng"),
        ],
        intent_coverage="model_proposed",
    )

    # Paraphrased note stays proposed (confirmed=False)
    _, response = run_turn(
        runtime, history, "Ghi lại kết quả",
        [
            action("record_work_note", text="Kết quả đo đạc khoảng bốn mươi hai", source_result=ref),
            answer("Đã ghi"),
        ],
        contract=contract,
    )

    assert response.reconciliation.declared_scope_completion == "unresolved"
    assert any(u["outcome_id"] == "out-note" and u["reason"] == "unconfirmed_proposal" for u in response.reconciliation.unmet_outcomes)

# 9. BOUNDARY REMOVAL: NOTE CREATED THEN REMOVED IN SAME EXECUTION IS UNMET
# =============================================================================

def test_note_created_then_removed_in_same_execution_fails_boundary_check():
    runtime, history = runtime_fixture(), ConversationHistory()
    runtime.artifact_store.put(Artifact("sample", "sample", 1, {"result": "42"}))
    ref = {"artifact_id": "sample", "revision": 1, "field": "result"}

    contract = TurnContract(
        outcomes=[OutcomeSpec("out-note", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note")],
        intent_coverage="model_proposed",
    )

    def remove_step():
        note_id = runtime.notebook.snapshot()[0]["id"]
        runtime.notebook.remove(note_id, revision=1, context=runtime.notebook.current_context())
        return answer("Đã ghi rồi xóa")

    def decisions_gen():
        yield action("record_work_note", text="42", source_result=ref)
        yield remove_step

    _, response = run_turn(runtime, history, "Ghi rồi xóa", decisions_gen(), contract=contract)

    assert response.reconciliation.declared_scope_completion == "unresolved"
    assert any(u["outcome_id"] == "out-note" and u["reason"] == "note_removed" for u in response.reconciliation.unmet_outcomes)

# 10. STALE REVISION UPDATE: TARGET REVISION MISMATCH IS UNMET
# =============================================================================

def test_stale_revision_update_is_unmet():
    runtime, history = runtime_fixture(), ConversationHistory()
    runtime.artifact_store.put(Artifact("sample", "sample", 1, {"result": "42"}))
    ref = {"artifact_id": "sample", "revision": 1, "field": "result"}

    # Establish note at revision 2
    runtime.notebook.begin_request("Setup")
    note1 = runtime.execute_tool("record_work_note", {"text": "42", "source_result": ref})["note"]
    runtime.execute_tool("record_work_note", {"text": "43", "source_result": ref, "replace_note_id": note1["id"], "revision": 1})

    # Obligation expects updating revision 2, but model sends expected_revision 1
    contract = TurnContract(
        outcomes=[
            OutcomeSpec("out-update", kind=OutcomeKind.REGISTERED_EFFECT, effect_type="record_work_note",
                        target_ref={"note_id": note1["id"], "expected_revision": 2}),
        ],
        intent_coverage="model_proposed",
    )

    _, response = run_turn(
        runtime, history, "Sửa ghi chú",
        [
            # Stale update attempt targeting revision 1
            action("record_work_note", text="44", source_result=ref, replace_note_id=note1["id"], revision=1),
            answer("Đã sửa"),
        ],
        contract=contract,
    )

    # The tool threw a ValueError for stale revision, so no commit happened
    assert response.reconciliation.declared_scope_completion == "unresolved"
