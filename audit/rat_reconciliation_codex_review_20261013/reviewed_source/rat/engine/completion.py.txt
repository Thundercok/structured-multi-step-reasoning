"""rat.engine.completion — Headless completion reconciliation and contract validation.

Implements ADR-003:
- Declared outcome contracts (frozen before first effect, immutable afterwards)
- Trusted execution journal with structured receipt envelopes
- Producer-consumer dependency matching across artifacts and notes
- Separation of lifecycle, declared-scope completion, intent coverage, and scoped proof quality
- Zero Qt dependencies. Pure functional reconciler.
"""
from __future__ import annotations

import copy
from dataclasses import asdict, dataclass, field
import hashlib
import json
import time
from typing import Any, Dict, List, Optional
from uuid import uuid4


class OutcomeKind:
    RESPONSE = "response"
    REGISTERED_EFFECT = "registered_effect"


@dataclass
class OutcomeSpec:
    """Declared expectation for a turn."""

    outcome_id: str
    kind: str = OutcomeKind.RESPONSE  # "response" | "registered_effect"
    effect_type: Optional[str] = None  # e.g. "calculate", "record_work_note", "select_work_results"
    description: str = ""
    target_ref: Optional[Dict[str, Any]] = None  # e.g. {"note_id": ..., "expected_revision": ...}
    source_ref: Optional[Dict[str, Any]] = None  # e.g. {"producer_outcome_id": ...} or {"artifact_id": ...}
    dependency_ids: List[str] = field(default_factory=list)
    allow_proposal: bool = False
    required_proof_scope: Optional[str] = None  # e.g. "source_calculation_note", "literal_arithmetic_answer"

    def to_dict(self) -> Dict[str, Any]:
        return {
            "outcome_id": self.outcome_id,
            "kind": self.kind,
            "effect_type": self.effect_type,
            "description": self.description,
            "target_ref": copy.deepcopy(self.target_ref) if self.target_ref is not None else None,
            "source_ref": copy.deepcopy(self.source_ref) if self.source_ref is not None else None,
            "dependency_ids": list(self.dependency_ids),
            "allow_proposal": self.allow_proposal,
            "required_proof_scope": self.required_proof_scope,
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> OutcomeSpec:
        return cls(
            outcome_id=str(data.get("outcome_id") or "out-" + uuid4().hex[:8]),
            kind=str(data.get("kind") or OutcomeKind.RESPONSE),
            effect_type=data.get("effect_type"),
            description=str(data.get("description") or ""),
            target_ref=copy.deepcopy(data.get("target_ref")),
            source_ref=copy.deepcopy(data.get("source_ref")),
            dependency_ids=list(data.get("dependency_ids") or []),
            allow_proposal=bool(data.get("allow_proposal", False)),
            required_proof_scope=data.get("required_proof_scope"),
        )


@dataclass
class TurnContract:
    """Pre-execution obligations. Frozen before the first side-effect and immutable thereafter."""

    schema_version: str = "1.0"
    contract_id: str = field(default_factory=lambda: "contract-" + uuid4().hex[:12])
    conversation_id: str = "conv-default"
    turn_id: str = "turn-default"
    execution_id: str = "exec-default"
    generation: Optional[int] = None
    proposed_by: str = "model"  # "model_planner" | "model_decision" | "fixture" | "unspecified"
    source_message_refs: List[str] = field(default_factory=list)
    outcomes: List[OutcomeSpec] = field(default_factory=list)
    intent_coverage: str = "model_proposed"  # "model_proposed" | "ambiguous" | "invalid" | "unchecked"
    frozen: bool = False

    def freeze(self) -> None:
        self.frozen = True

    def is_empty(self) -> bool:
        return len(self.outcomes) == 0

    def to_dict(self) -> Dict[str, Any]:
        return {
            "schema_version": self.schema_version,
            "contract_id": self.contract_id,
            "conversation_id": self.conversation_id,
            "turn_id": self.turn_id,
            "execution_id": self.execution_id,
            "generation": self.generation,
            "proposed_by": self.proposed_by,
            "source_message_refs": list(self.source_message_refs),
            "outcomes": [o.to_dict() for o in self.outcomes],
            "intent_coverage": self.intent_coverage,
            "frozen": self.frozen,
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> TurnContract:
        outcomes_raw = data.get("outcomes") or []
        outcomes = [OutcomeSpec.from_dict(o) if isinstance(o, dict) else o for o in outcomes_raw]
        return cls(
            schema_version=str(data.get("schema_version") or "1.0"),
            contract_id=str(data.get("contract_id") or "contract-" + uuid4().hex[:12]),
            conversation_id=str(data.get("conversation_id") or "conv-default"),
            turn_id=str(data.get("turn_id") or "turn-default"),
            execution_id=str(data.get("execution_id") or "exec-default"),
            generation=data.get("generation"),
            proposed_by=str(data.get("proposed_by") or "model"),
            source_message_refs=list(data.get("source_message_refs") or []),
            outcomes=outcomes,
            intent_coverage=str(data.get("intent_coverage") or "model_proposed"),
            frozen=bool(data.get("frozen", False)),
        )


@dataclass
class ReceiptEnvelope:
    """Authoritative execution receipt emitted by runtime components."""

    receipt_id: str = field(default_factory=lambda: "rcpt-" + uuid4().hex[:12])
    issuer: str = "runtime"  # "dispatcher" | "notebook" | "verification" | "artifact_store"
    sequence: int = 0
    conversation_id: str = ""
    turn_id: str = ""
    execution_id: str = ""
    generation: Optional[int] = None
    operation_call_id: Optional[str] = None
    status: str = "committed"  # "committed" | "error" | "cancelled" | "needs_input"
    committed: bool = True
    operation_name: Optional[str] = None
    operation_args: Optional[Dict[str, Any]] = None
    operation_result: Any = None
    artifact_refs: List[Dict[str, Any]] = field(default_factory=list)
    note_event: Optional[Dict[str, Any]] = None
    verification_report: Optional[Dict[str, Any]] = None
    before_revision: Optional[int] = None
    after_revision: Optional[int] = None
    parent_receipt_ids: List[str] = field(default_factory=list)
    evidence_hash: Optional[str] = None
    error: Optional[str] = None
    timestamp: float = field(default_factory=time.time)

    def to_dict(self) -> Dict[str, Any]:
        return {
            "receipt_id": self.receipt_id,
            "issuer": self.issuer,
            "sequence": self.sequence,
            "conversation_id": self.conversation_id,
            "turn_id": self.turn_id,
            "execution_id": self.execution_id,
            "generation": self.generation,
            "operation_call_id": self.operation_call_id,
            "status": self.status,
            "committed": self.committed,
            "operation_name": self.operation_name,
            "operation_args": copy.deepcopy(self.operation_args) if isinstance(self.operation_args, dict) else self.operation_args,
            "operation_result": copy.deepcopy(self.operation_result) if isinstance(self.operation_result, (dict, list)) else self.operation_result,
            "artifact_refs": copy.deepcopy(self.artifact_refs),
            "note_event": copy.deepcopy(self.note_event) if self.note_event is not None else None,
            "verification_report": copy.deepcopy(self.verification_report) if self.verification_report is not None else None,
            "before_revision": self.before_revision,
            "after_revision": self.after_revision,
            "parent_receipt_ids": list(self.parent_receipt_ids),
            "evidence_hash": self.evidence_hash,
            "error": self.error,
            "timestamp": self.timestamp,
        }


@dataclass
class ExecutionJournal:
    """Request-scoped trusted journal recording all runtime events and receipts."""

    conversation_id: str
    turn_id: str
    execution_id: str
    generation: Optional[int] = None
    receipts: List[ReceiptEnvelope] = field(default_factory=list)
    _seq: int = 0

    def record_operation(
        self,
        operation_name: str,
        args: Dict[str, Any],
        result: Any,
        *,
        status: str = "committed",
        committed: bool = True,
        artifact_refs: Optional[List[Dict[str, Any]]] = None,
        error: Optional[str] = None,
        operation_call_id: Optional[str] = None,
    ) -> ReceiptEnvelope:
        self._seq += 1
        digest = None
        try:
            raw = json.dumps({"op": operation_name, "args": args, "result": result}, sort_keys=True, default=str)
            digest = hashlib.sha256(raw.encode()).hexdigest()
        except Exception:
            pass

        rcpt = ReceiptEnvelope(
            issuer="dispatcher",
            sequence=self._seq,
            conversation_id=self.conversation_id,
            turn_id=self.turn_id,
            execution_id=self.execution_id,
            generation=self.generation,
            operation_call_id=operation_call_id or f"call-{self._seq}",
            status=status,
            committed=committed,
            operation_name=operation_name,
            operation_args=args,
            operation_result=result,
            artifact_refs=list(artifact_refs or []),
            evidence_hash=digest,
            error=error,
        )
        self.receipts.append(rcpt)
        return rcpt

    def record_note_event(self, event_dict: Dict[str, Any], *, operation_receipt_id: Optional[str] = None) -> ReceiptEnvelope:
        self._seq += 1
        rcpt = ReceiptEnvelope(
            issuer="notebook",
            sequence=self._seq,
            conversation_id=self.conversation_id,
            turn_id=self.turn_id,
            execution_id=self.execution_id,
            generation=self.generation,
            status="committed",
            committed=True,
            operation_name="record_work_note",
            note_event=copy.deepcopy(event_dict),
            before_revision=event_dict.get("before_revision"),
            after_revision=event_dict.get("after_revision"),
            parent_receipt_ids=[operation_receipt_id] if operation_receipt_id else [],
        )
        self.receipts.append(rcpt)
        return rcpt

    def record_verification(self, report_dict: Dict[str, Any], *, operation_receipt_id: Optional[str] = None) -> ReceiptEnvelope:
        self._seq += 1
        rcpt = ReceiptEnvelope(
            issuer="verification",
            sequence=self._seq,
            conversation_id=self.conversation_id,
            turn_id=self.turn_id,
            execution_id=self.execution_id,
            generation=self.generation,
            status="committed",
            committed=True,
            verification_report=copy.deepcopy(report_dict),
            parent_receipt_ids=[operation_receipt_id] if operation_receipt_id else [],
        )
        self.receipts.append(rcpt)
        return rcpt

    def to_dict(self) -> Dict[str, Any]:
        return {
            "conversation_id": self.conversation_id,
            "turn_id": self.turn_id,
            "execution_id": self.execution_id,
            "generation": self.generation,
            "receipts": [r.to_dict() for r in self.receipts],
        }


@dataclass
class TurnReconciliation:
    """The honest truth of turn completion based on actual receipts."""

    contract_id: Optional[str] = None
    lifecycle: str = "completed"  # "completed" | "in_progress" | "needs_input" | "failed" | "cancelled"
    intent_coverage: str = "model_proposed"  # "model_proposed" | "ambiguous" | "invalid" | "unchecked"
    declared_scope_completion: str = "satisfied"  # "satisfied" | "partial" | "unresolved" | "not_applicable"
    scoped_quality: str = "not_checked"  # "PASS" | "FAIL" | "UNKNOWN" | "not_checked"
    matched_receipts: List[str] = field(default_factory=list)
    unmet_outcomes: List[Dict[str, Any]] = field(default_factory=list)
    unresolved_reasons: List[str] = field(default_factory=list)
    semantic_limitation: str = "Receipt matching validates recorded effects and scoped proofs, but does not prove unverified prose assertions."
    summary: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return {
            "contract_id": self.contract_id,
            "lifecycle": self.lifecycle,
            "intent_coverage": self.intent_coverage,
            "declared_scope_completion": self.declared_scope_completion,
            "scoped_quality": self.scoped_quality,
            "matched_receipts": list(self.matched_receipts),
            "unmet_outcomes": copy.deepcopy(self.unmet_outcomes),
            "unresolved_reasons": list(self.unresolved_reasons),
            "semantic_limitation": self.semantic_limitation,
            "summary": self.summary,
        }


def reconcile_turn(
    contract: Optional[TurnContract],
    journal: ExecutionJournal,
    *,
    lifecycle: str,
    final_answer: str = "",
    error: Optional[str] = None,
    notebook_snapshot: Optional[List[Dict[str, Any]]] = None,
    store: Optional[Any] = None,
) -> TurnReconciliation:
    """Pure functional reconciliation matching frozen obligations against trusted journal receipts."""
    # 1. Determine scoped quality from verification reports in journal
    scoped_quality = "not_checked"
    verif_reports = [r.verification_report for r in journal.receipts if r.verification_report]
    if verif_reports:
        if any(rep.get("status") == "FAIL" for rep in verif_reports):
            scoped_quality = "FAIL"
        elif any(rep.get("status") == "UNKNOWN" for rep in verif_reports):
            scoped_quality = "UNKNOWN"
        elif all(rep.get("status") == "PASS" for rep in verif_reports):
            scoped_quality = "PASS"

    # 2. Handle missing or unchecked contract
    if contract is None or contract.intent_coverage in ("unchecked", "invalid"):
        intent_cov = contract.intent_coverage if contract else "unchecked"
        if lifecycle == "cancelled":
            return TurnReconciliation(
                contract_id=contract.contract_id if contract else None,
                lifecycle="cancelled",
                intent_coverage=intent_cov,
                declared_scope_completion="unresolved",
                scoped_quality=scoped_quality,
                unresolved_reasons=["cancelled_in_flight"],
                summary="Lượt bị dừng trước khi hoàn tất.",
            )
        if lifecycle == "needs_input":
            return TurnReconciliation(
                contract_id=contract.contract_id if contract else None,
                lifecycle="needs_input",
                intent_coverage=intent_cov,
                declared_scope_completion="unresolved",
                scoped_quality=scoped_quality,
                unresolved_reasons=["pending_input_required"],
                summary="Đang chờ thông tin bổ sung từ người dùng.",
            )
        if lifecycle == "failed" or error:
            return TurnReconciliation(
                contract_id=contract.contract_id if contract else None,
                lifecycle="failed",
                intent_coverage=intent_cov,
                declared_scope_completion="unresolved",
                scoped_quality=scoped_quality,
                unresolved_reasons=[error or "execution_failed"],
                summary=f"Thực thi gặp lỗi: {error}",
            )
        # Normal conversational reply with no declared contract
        return TurnReconciliation(
            contract_id=contract.contract_id if contract else None,
            lifecycle="completed",
            intent_coverage=intent_cov,
            declared_scope_completion="unchecked",
            scoped_quality=scoped_quality,
            summary="Câu trả lời hội thoại (chưa khai hợp đồng công việc cụ thể).",
        )

    # 3. Contract is present. Handle non-normal lifecycles
    if lifecycle == "cancelled":
        unmet = [
            {"outcome_id": o.outcome_id, "reason": "cancelled", "description": o.description}
            for o in contract.outcomes
        ]
        return TurnReconciliation(
            contract_id=contract.contract_id,
            lifecycle="cancelled",
            intent_coverage=contract.intent_coverage,
            declared_scope_completion="unresolved",
            scoped_quality=scoped_quality,
            unmet_outcomes=unmet,
            unresolved_reasons=["cancelled_in_flight"],
            summary="Lượt bị dừng; các công việc đã khai chưa hoàn thành.",
        )

    if lifecycle == "needs_input":
        unmet = [
            {"outcome_id": o.outcome_id, "reason": "needs_input", "description": o.description}
            for o in contract.outcomes
        ]
        return TurnReconciliation(
            contract_id=contract.contract_id,
            lifecycle="needs_input",
            intent_coverage=contract.intent_coverage,
            declared_scope_completion="unresolved",
            scoped_quality=scoped_quality,
            unmet_outcomes=unmet,
            unresolved_reasons=["pending_input_required"],
            summary="Đang chờ thông tin bổ sung; các công việc giữ trạng thái chờ.",
        )

    if lifecycle == "failed" or error:
        unmet = [
            {"outcome_id": o.outcome_id, "reason": "execution_failed", "description": o.description}
            for o in contract.outcomes
        ]
        return TurnReconciliation(
            contract_id=contract.contract_id,
            lifecycle="failed",
            intent_coverage=contract.intent_coverage,
            declared_scope_completion="unresolved",
            scoped_quality=scoped_quality,
            unmet_outcomes=unmet,
            unresolved_reasons=[error or "execution_failed"],
            summary=f"Lỗi thực thi: {error}",
        )

    # 4. Normal boundary: evaluate each outcome against receipts
    matched_receipts: List[str] = []
    unmet_outcomes: List[Dict[str, Any]] = []
    producer_artifacts: Dict[str, Dict[str, Any]] = {}  # outcome_id -> artifact ref

    # Filter receipts belonging strictly to the current execution
    current_receipts = [r for r in journal.receipts if r.execution_id == contract.execution_id]

    for outcome in contract.outcomes:
        if outcome.kind == OutcomeKind.RESPONSE:
            if final_answer.strip():
                matched_receipts.append(f"response:{contract.turn_id}")
            else:
                unmet_outcomes.append({
                    "outcome_id": outcome.outcome_id,
                    "reason": "missing_response",
                    "description": outcome.description,
                })
            continue

        # Check explicit dependency_ids
        if outcome.dependency_ids:
            unmet_deps = [dep for dep in outcome.dependency_ids if any(u["outcome_id"] == dep for u in unmet_outcomes)]
            if unmet_deps:
                unmet_outcomes.append({
                    "outcome_id": outcome.outcome_id,
                    "reason": "dependency_unmet",
                    "description": f"Phụ thuộc {unmet_deps} chưa hoàn thành",
                })
                continue

        if outcome.kind == OutcomeKind.REGISTERED_EFFECT:
            # Effect 1: Calculate
            if outcome.effect_type == "calculate":
                calc_receipts = [
                    r for r in current_receipts
                    if r.operation_name == "calculate" and r.committed and r.status == "committed"
                ]
                if not calc_receipts:
                    unmet_outcomes.append({
                        "outcome_id": outcome.outcome_id,
                        "reason": "omitted_action",
                        "description": outcome.description or "Chưa thực hiện phép tính",
                    })
                    continue

                calc_rcpt = calc_receipts[0]
                # Check required proof
                if outcome.required_proof_scope:
                    matching_reports = [
                        rep for rep in verif_reports
                        if rep.get("scope") == outcome.required_proof_scope
                    ]
                    if not matching_reports or matching_reports[-1].get("status") != "PASS":
                        unmet_outcomes.append({
                            "outcome_id": outcome.outcome_id,
                            "reason": "proof_failed",
                            "description": "Proof số học bắt buộc không đạt",
                        })
                        continue

                # Record producer artifact
                if calc_rcpt.artifact_refs:
                    producer_artifacts[outcome.outcome_id] = calc_rcpt.artifact_refs[0]
                matched_receipts.append(calc_rcpt.receipt_id)
                continue

            # Effect 2: Record/Update work note
            if outcome.effect_type == "record_work_note":
                note_receipts = [
                    r for r in current_receipts
                    if r.issuer == "notebook" and r.note_event and r.committed
                ]
                if not note_receipts:
                    unmet_outcomes.append({
                        "outcome_id": outcome.outcome_id,
                        "reason": "omitted_action",
                        "description": outcome.description or "Chưa ghi nhận thao tác lưu/sửa sổ tay",
                    })
                    continue

                # Find candidate note event matching this outcome
                matched_event = None
                for n_rcpt in note_receipts:
                    evt = n_rcpt.note_event
                    if not evt:
                        continue
                    snapshot = evt.get("snapshot") or {}
                    source_ref = snapshot.get("source_ref") or {}

                    # A. Producer-consumer linkage check
                    if outcome.source_ref:
                        producer_id = outcome.source_ref.get("producer_outcome_id")
                        if producer_id:
                            expected_prod = producer_artifacts.get(producer_id)
                            if not expected_prod:
                                # Producer failed or did not emit artifact
                                continue
                            if source_ref.get("artifact_id") != expected_prod.get("artifact_id") or \
                               source_ref.get("artifact_revision") != expected_prod.get("revision"):
                                continue

                        # Specific artifact ref check
                        if "artifact_id" in outcome.source_ref:
                            if source_ref.get("artifact_id") != outcome.source_ref["artifact_id"]:
                                continue
                            if "revision" in outcome.source_ref and source_ref.get("artifact_revision") != outcome.source_ref["revision"]:
                                continue

                    # B. Target note update check
                    if outcome.target_ref:
                        target_id = outcome.target_ref.get("note_id")
                        if target_id and evt.get("note_id") != target_id:
                            continue
                        expected_rev = outcome.target_ref.get("expected_revision")
                        if expected_rev is not None and evt.get("before_revision") != expected_rev:
                            continue

                    # C. Allow proposal check
                    if not outcome.allow_proposal and not snapshot.get("confirmed", False):
                        continue

                    # D. Required proof scope check
                    if outcome.required_proof_scope:
                        verif_info = snapshot.get("source_verification") or {}
                        if verif_info.get("status") != "PASS" or verif_info.get("scope") != outcome.required_proof_scope:
                            continue

                    matched_event = n_rcpt
                    break

                if matched_event is None:
                    # Provide specific failure diagnostic
                    last_note = note_receipts[-1].note_event.get("snapshot") or {}
                    reason = "wrong_source"
                    if outcome.source_ref and outcome.source_ref.get("producer_outcome_id"):
                        reason = "wrong_source_consumer"
                    elif not outcome.allow_proposal and not last_note.get("confirmed", False):
                        reason = "unconfirmed_proposal"
                    elif outcome.required_proof_scope:
                        reason = "proof_failed"
                    unmet_outcomes.append({
                        "outcome_id": outcome.outcome_id,
                        "reason": reason,
                        "description": outcome.description or "Ghi chú không khớp nguồn/ràng buộc yêu cầu",
                    })
                    continue

                # E. Boundary check: check if the note is active at completion boundary
                target_note_id = matched_event.note_event.get("note_id")
                is_removed_in_receipts = any(
                    r.note_event and r.note_event.get("kind") == "remove" and r.note_event.get("note_id") == target_note_id
                    for r in current_receipts
                )
                is_missing_from_snapshot = (notebook_snapshot is not None and not any(n.get("id") == target_note_id for n in notebook_snapshot))
                if is_removed_in_receipts or is_missing_from_snapshot:
                    unmet_outcomes.append({
                        "outcome_id": outcome.outcome_id,
                        "reason": "note_removed",
                        "description": "Ghi chú đã bị xóa trước khi kết thúc lượt",
                    })
                    continue

                matched_receipts.append(matched_event.receipt_id)
                continue

            # Generic registered effect (e.g. select_work_results, inspect_artifact)
            generic_receipts = [
                r for r in current_receipts
                if r.operation_name == outcome.effect_type and r.committed and r.status == "committed"
            ]
            if generic_receipts:
                matched_receipts.append(generic_receipts[0].receipt_id)
            else:
                unmet_outcomes.append({
                    "outcome_id": outcome.outcome_id,
                    "reason": "omitted_action",
                    "description": outcome.description or f"Chưa gọi thao tác {outcome.effect_type}",
                })

    # 5. Determine overall declared-scope completion
    effect_outcomes = [o for o in contract.outcomes if o.kind == OutcomeKind.REGISTERED_EFFECT]
    unresolved_reasons = [u["reason"] for u in unmet_outcomes]

    if not effect_outcomes:
        completion = "satisfied" if not unmet_outcomes else "unresolved"
        summary = "Đã hoàn thành phản hồi hội thoại." if not unmet_outcomes else "Chưa hoàn tất phản hồi."
    else:
        unmet_effect_ids = {u["outcome_id"] for u in unmet_outcomes}
        met_effect_count = sum(1 for o in effect_outcomes if o.outcome_id not in unmet_effect_ids)

        if met_effect_count == len(effect_outcomes):
            completion = "satisfied"
            summary = "Đã hoàn thành đầy đủ các công việc đã khai và đối chiếu biên nhận."
        elif met_effect_count > 0:
            completion = "partial"
            summary = "Chỉ hoàn thành một phần công việc; còn thiếu một số thao tác/ràng buộc."
        else:
            completion = "unresolved"
            summary = "Chưa hoàn tất công việc yêu cầu; thiếu biên nhận thực thi."

    return TurnReconciliation(
        contract_id=contract.contract_id,
        lifecycle="completed",
        intent_coverage=contract.intent_coverage,
        declared_scope_completion=completion,
        scoped_quality=scoped_quality,
        matched_receipts=matched_receipts,
        unmet_outcomes=unmet_outcomes,
        unresolved_reasons=unresolved_reasons,
        summary=summary,
    )
