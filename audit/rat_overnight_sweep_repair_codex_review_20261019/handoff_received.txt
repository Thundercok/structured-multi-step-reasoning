def make_candidate_label(c, all_candidates):
    name = c.get("name", "")
    p = c.get("path")
    if p:
        try:
            all_paths = [cand.get("path") for cand in all_candidates if cand.get("path")]
            if len(all_paths) > 1:
                cp = os.path.commonpath(all_paths)
                rel = os.path.relpath(p, cp)
                # If rel ends with app name, show parent folder
                parent_folder = Path(p).parent.name
                diff = parent_folder if parent_folder else rel
            else:
                diff = Path(p).parent.name
            return f"{diff} · {name}"
        except Exception:
            return f"{Path(p).parent.name} · {name}"
    return name

candidates = [dict(name="League of Legends", path=p) for p in paths]
visible_labels = []
for c in candidates:
    btn = QPushButton(make_candidate_label(c, candidates))
    btn.resize(131, 30)
    option = QStyleOptionButton()
    btn.initStyleOption(option)
    contents = btn.style().subElementRect(QStyle.SubElement.SE_PushButtonContents, option, btn)
    text, fm = btn.text(), btn.fontMetrics()
    prefix = ""
    for char in text:
        if fm.horizontalAdvance(prefix + char) > contents.width():
            break
        prefix += char
    visible_labels.append(prefix)
    print("Full text:", text, "Contents width:", contents.width(), "Visible prefix:", repr(prefix))

print("Set of visible prefixes:", set(visible_labels))
print("Distinguishable?", len(set(visible_labels)) == 2)
