    trailing = TRAILING_NUMERIC_RESULT.search(text)
    if trailing:
        return extract_number(trailing.group(1))
    if NUMERIC_ANSWER.fullmatch(text) or any(cue in lowered for cue in conclusion_cues):

def normalize_answer(ans: str) -> str:
    s = (ans or "").strip()
    while True:
        before = s
        if s.startswith("**") and s.endswith("**") and len(s) >= 4:
            s = s[2:-2].strip()
        m = re.match(
            r"^(?:#{1,6}\s*)?(?:[^\w\s]+\s*)?(?:\*\*)?(?:final\s+)?answer(?:\*\*)?\s*:\s*",
            s,
            flags=re.IGNORECASE,
        )
        if m:
            s = s[m.end():].strip()
        if s.endswith(".") and not re.search(r"\d\.\d", s):
            s = s[:-1].strip()
        if s.startswith("**") and s.endswith("**") and len(s) >= 4:
            s = s[2:-2].strip()
        if s == before:
            break
    return s

def _clean_cand(norm: str) -> str:
    yes_no = YES_NO_ANSWER.match(norm)
    if yes_no:
        return yes_no.group(1)
    for p in placeholders:
        if norm.casefold().startswith(p.casefold()):
            norm = re.sub(r"^\s*" + re.escape(p) + r"\s*[:=]?\s*", "", norm, flags=re.IGNORECASE).strip()
            break
    norm = norm.lstrip("=: ").strip()
    equals_results = EQUALS_RESULT.findall(norm)
    if equals_results:
        return extract_number(equals_results[-1])
    trailing = TRAILING_NUMERIC_RESULT.search(norm)
    if trailing:
        return extract_number(trailing.group(1))
    rhs = norm.rsplit("=", 1)[-1].strip() if "=" in norm else norm
    is_expr = any(op in rhs for op in ("+", "*", "/", "×", "÷")) or ("-" in rhs and not re.match(r"^-\s*\d", rhs.strip()))
    if not is_expr:
        num = extract_conclusion_number(norm)
        if num is not None:
            return num
    return norm

def extract_answer(text: str) -> str:
    lines = text.splitlines()
    marker_re = re.compile(
        r"(?:#{1,6}\s*)?(?:[^\w\s]+\s*)?(?:\*\*)?(?:final\s+)?answer(?:\*\*)?\s*:\s*(.*)",
        re.IGNORECASE,
    )
    ans = ""
    for i in range(len(lines) - 1, -1, -1):
        line = lines[i].strip()
        m = marker_re.search(line)
        if m:
            val = m.group(1).strip()
            if line.startswith("**") and line.endswith("**") and len(line) >= 4:
                val = line
            if val:
                norm = normalize_answer(val)
                if norm and norm.casefold() not in placeholders:
                    ans = _clean_cand(norm)
                    break
            for j in range(i + 1, len(lines)):
                next_l = lines[j].strip()
                if next_l and not next_l.startswith("```"):
                    norm = normalize_answer(next_l)
                    if norm and norm.casefold() not in placeholders:
                        ans = _clean_cand(norm)
                        break
            if ans:
                break
    if not ans:
        for l in reversed(lines):
            l = l.strip()
            if not l or l.startswith("```") or l.casefold() in placeholders:
                continue
            yes_no = YES_NO_ANSWER.match(l)
            if yes_no:
                ans = yes_no.group(1)
                break
            norm = normalize_answer(l)
            if norm and norm.casefold() not in placeholders:
                ans = _clean_cand(norm)
                break
    return ans
