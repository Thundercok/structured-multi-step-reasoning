import re
from fractions import Fraction

NUM = re.compile(r'-?\d[\d,]*\.?\d*')
NUMERIC_ANSWER = re.compile(r'^\s*-?\d[\d,]*\.?\d*(?:\s+[^\d]*)?\s*$')
EQUALS_RESULT = re.compile(r'=\s*(-?(?:\d[\d,]*(?:\.\d+)?|\.\d+))(?!\.?\d)(?!\s*[/÷×*+-])', re.IGNORECASE)
TRAILING_NUMERIC_RESULT = re.compile(r'[:=]\s*(-?(?:\d[\d,]*(?:\.\d+)?|\.\d+))(?!\.?\d)(?!\s*[/÷×*+-])(?:\s+[^\d\+\-\*\/\(\)\=]+)?\.?\s*$', re.IGNORECASE)
YES_NO_ANSWER = re.compile(r'^\s*(yes|no)\b', re.IGNORECASE)

def extract_number(text: str) -> str | None:
    nums = NUM.findall(text.replace(',', ''))
    if not nums:
        return None
    val = nums[-1]
    if val.endswith('.') and not re.search(r'\d\.\d', val):
        val = val.rstrip('.')
    return val

def extract_conclusion_number(text: str) -> str | None:
    lowered = text.casefold()
    conclusion_cues = (
        'therefore', 'thus', 'hence', 'final answer', 'kết quả', 'đáp án',
        'do đó', 'vậy', 'total',
    )
    clauses = [c.strip() for c in re.split(r'[;\n]', text) if c.strip()]
    target = clauses[-1] if clauses else text
    rhs = target.rsplit('=', 1)[-1].strip() if '=' in target else target
    if any(op in rhs for op in ('/', '*', '+', '÷', '×')) or ('-' in rhs and not re.match(r'^-\s*\d', rhs.strip())):
        return None
    equals_results = EQUALS_RESULT.findall(target)
    if equals_results:
        return extract_number(equals_results[-1])
    trailing = TRAILING_NUMERIC_RESULT.search(target)
    if trailing:
        return extract_number(trailing.group(1))
    if NUMERIC_ANSWER.fullmatch(target) or any(cue in target.casefold() for cue in conclusion_cues) or any(cue in lowered for cue in conclusion_cues):

    return None
