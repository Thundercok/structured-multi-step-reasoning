def test_fn(norm):
    placeholders = {"<kết quả>", "<your final answer>", "[kết quả của bạn]", "<final answer>", "<answer>"}
    for p in placeholders:
        if norm.casefold().startswith(p.casefold()):
            norm = re.sub(r"^\s*" + re.escape(p) + r"\s*[:=]?\s*", "", norm, flags=re.IGNORECASE).strip()
            break
    norm = norm.lstrip("=: ").strip()
    m_term = re.search(r"=\s*(-?\d[\d,]*\.?\d*)(?!\s*[/÷×*+-])(?:\s+[^\d\+\-\*\/\(\)\=]+)?\.?$", norm)
    if m_term:
        val = m_term.group(1).replace(",", "")
        if val.endswith(".") and not re.search(r"\d\.\d", val):
            val = val.rstrip(".")
        return val
    return norm
