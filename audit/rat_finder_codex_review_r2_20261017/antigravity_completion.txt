from PyQt6.QtWidgets import QApplication, QWidget, QHBoxLayout, QVBoxLayout, QLabel, QPushButton, QSizePolicy
from PyQt6.QtCore import Qt

class ElidedLabel(QLabel):
    def __init__(self, text: str = "", parent = None, elide_mode = Qt.TextElideMode.ElideRight):
        super().__init__(parent)
        self.full_text = str(text or "")
        self.elide_mode = elide_mode
        self.setTextFormat(Qt.TextFormat.PlainText)
        self.setToolTip(self.full_text)
        self.setMinimumWidth(0)
        self.setSizePolicy(QSizePolicy.Policy.Ignored, QSizePolicy.Policy.Preferred)
        self.setText(self.full_text)

    def resizeEvent(self, event):
        super().resizeEvent(event)
        self._update_elided()

    def _update_elided(self):
        w = max(0, self.width())
        if w > 0 and self.full_text:
            super().setText(self.fontMetrics().elidedText(self.full_text, self.elide_mode, w))
        else:
            super().setText(self.full_text)
