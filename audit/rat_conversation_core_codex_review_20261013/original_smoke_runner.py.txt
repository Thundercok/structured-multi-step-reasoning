"""Headless real-model development smoke check for the shared conversation core using the 6-turn DSA example."""
import json
import tempfile
from pathlib import Path
from rat.crawler.db import Database
from rat.engine.assistant_runtime import AssistantRuntime
from rat.timetable.compositor import TimetableCompositor
from rat.ui.chat_session import ConversationHistory


def main():
    out_dir = Path("/Users/thundercock2/Documents/Github/multi-step-structured-reasoning/audit/rat_conversation_core_20261013")
    out_dir.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory() as tmp:
        db = Database(str(Path(tmp) / "smoke.db"))
        runtime = AssistantRuntime(db=db, compositor=TimetableCompositor([]), use_live_model=True)
        history = ConversationHistory()

        turns_spec = [
            ("turn_1", "Giải thích binary search bằng ví dụ nhỏ", False),
            ("turn_2", "Dùng mảng [2, 4, 6, 8, 10], tìm 8 từng bước và ghi chú lại mảng cùng mục tiêu tìm 8 giúp tôi", False),
            ("turn_3", "Vì sao bước đó dùng mid + 1 mà không giữ mid?", False),
            ("turn_4", "Đổi thành tìm 6 thì các bước đổi thế nào?", False),
            ("turn_5", "Quay lại câu hỏi mid + 1 ở ví dụ trước, viết gọn thành 3 ý để tôi chép bài", False),
            ("turn_6_cancelled", "Viết bảng trace đầy đủ cho mọi trường hợp", True),
        ]

        records = []
        for label, prompt, cancel_immediately in turns_spec:
            snap = history.begin_turn(prompt)
            prior_messages = history.messages_with_turn_ids()[:-1]
            prior_refs = history.turn_references()[:-1]
            resp = runtime.run(
                prompt,
                max_steps=4,
                history=prior_messages,
                turn_ref=snap.to_turn_ref(history.conversation_id),
                turn_references=prior_refs,
                should_cancel=(lambda: True) if cancel_immediately else None,
            )
            if resp.error == "cancelled":
                history.cancel_turn(snap.turn_id, execution_id=resp.execution_id)
            elif resp.error:
                history.fail_turn(snap.turn_id, error=resp.error, execution_id=resp.execution_id)
            else:
                history.complete_turn(
                    snap.turn_id,
                    resp.answer,
                    execution_id=resp.execution_id,
                    note_ids=[n["id"] for n in runtime.notebook.notes_for_turn(snap.turn_id)],
                )
            records.append({
                "label": label,
                "turn_id": snap.turn_id,
                "execution_id": resp.execution_id,
                "prompt": prompt,
                "is_live_model": resp.is_live_model,
                "answer": resp.answer,
                "error": resp.error,
                "prompt_tokens": resp.prompt_tokens,
                "completion_tokens": resp.completion_tokens,
                "total_tokens": resp.total_tokens,
                "model_time_ms": round(resp.model_time_ms, 2),
                "tool_time_ms": round(resp.tool_time_ms, 2),
                "total_time_ms": round(resp.total_time_ms, 2),
                "tool_call_count": resp.tool_call_count,
                "note_events": resp.note_events,
                "turn_notes": runtime.notebook.notes_for_turn(snap.turn_id),
                "active_notes_snapshot": runtime.notebook.snapshot(),
                "steps": [
                    {
                        "step_number": s.step_number,
                        "tool_name": s.tool_name,
                        "tool_args": s.tool_args,
                        "tool_result": s.tool_result,
                        "error": s.error,
                        "prompt_tokens": s.prompt_tokens,
                        "completion_tokens": s.completion_tokens,
                        "raw_completion": s.raw_completion,
                    }
                    for s in resp.steps
                ],
            })

        report = {
            "scope": "development_smoke_not_benchmark_or_stage0_gate",
            "conversation_id": history.conversation_id,
            "turns": records,
            "final_active_notes": runtime.notebook.snapshot(),
            "historical_replaced_notes": runtime.notebook.historical_context(),
            "turn_references": history.turn_references(),
        }
        out_path = out_dir / "dsa_core_smoke.json"
        out_path.write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
        print(f"Wrote {out_path}")


if __name__ == "__main__":
    main()
